function lengthOfLongestSubstring(s: string): number {
    if (s.length < 2) {
        return s.length;
    }
    let longestLength = 1;
    const currentSet = new Set<string>();
    for (let i = s.length; i-- > 1;) {
        currentSet.add(s[i]);
        for (let j = i; j--;) {
            if (currentSet.has(s[j])) {
                if (longestLength < currentSet.size) {
                   longestLength = currentSet.size;
                }
                currentSet.clear();
                break;
            }
            currentSet.add(s[j]);
        }
    }
    return Math.max(longestLength, currentSet.size);
};
